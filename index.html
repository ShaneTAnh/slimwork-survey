function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var d = JSON.parse(e.postData.contents);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['Mã HS','Thời gian','Họ tên','Bộ phận','Đang dùng AI','Hiện trạng AI (Mục 2)','Công cụ AI (Mục 2.1)','Mức độ (0-5)','Danh hiệu','Ý tưởng Tự động hóa (4A)','Khả thi (4B)','Hình thức đào tạo (Mục 5)','Chủ đề hỗ trợ (Mục 6)']);
  }
  sheet.appendRow([
    d.id, d.timestamp, d.name, d.department, d.usingAI ? 'Có' : 'Chưa',
    (d.usageContexts || []).join('; '), (d.aiTools21 || []).join('; '),
    d.level, d.tier, d.automationIdea, d.feasibilityLabel,
    (d.trainingFormats || []).join('; '), (d.supportTopics || []).join('; ')
  ]);
  return ContentService.createTextOutput(JSON.stringify({status:'ok'})).setMimeType(ContentService.MimeType.JSON);
}
